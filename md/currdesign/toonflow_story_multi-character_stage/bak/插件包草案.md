## 4. entry.js 骨架

```javascript
// toonflow_story_multi_character_stage - entry.js

const NS = 'mcs';

let orchestrating = false;
let pendingSpeakers = [];

// ========== Config ==========
function getConfig() {
  const get = (k, fallback) => {
    const v = tavo.plugin.config.get(k);
    return v !== undefined ? v : fallback;
  };
  return {
    enabled: get('enabled', true) !== false,
    npcAutoTurn: get('npcAutoTurn', true) === true,
    npcAutoInterval: Number(get('npcAutoInterval', 3)) || 3,
    memoryIntegration: get('memoryIntegration', true) === true,
    speakerMode: get('speakerMode', 'auto'),
  };
}

// ========== Default State ==========
function defaultState() {
  return {
    version: 1,
    config: {
      enabled: true,
      npcAutoTurn: true,
      npcAutoInterval: 3,
    },
    context: {
      currentSpeaker: null,
      areaContext: {
        mainArea: '默认场景',
        activeRoles: [],
        remoteRoles: [],
        exceptionRoles: [],
      },
      dialogueHistory: [],
      turnsSinceUserInput: 0,
      pendingSpeakers: [],
    },
    characters: {},
  };
}

// ========== Speaker Route Engine ==========
function decideSpeaker(ctx) {
  const { recentDialogue, areaContext, turnsSinceUserInput, config } = ctx;
  const lastMsg = recentDialogue[recentDialogue.length - 1];

  // 规则路由
  // 1. 如果有 pending speakers，先处理
  if (pendingSpeakers.length > 0) {
    return { ...pendingSpeakers.shift(), mode: 'sequential' };
  }

  // 2. 如果用户刚发言，下一轮是 NPC 回应
  if (lastMsg?.isUserMessage) {
    // 选择一个合理的 NPC 回应
    const activeNpcs = areaContext.activeRoles.filter(r => r !== '用户');
    if (activeNpcs.length > 0) {
      return {
        role: activeNpcs[0],
        motive: '回应用户的发言',
        mode: 'normal',
      };
    }
  }

  // 3. NPC 自主发言
  if (config.npcAutoTurn && turnsSinceUserInput >= config.npcAutoInterval) {
    const activeNpcs = areaContext.activeRoles.filter(r => r !== '用户');
    if (activeNpcs.length > 0) {
      return {
        role: activeNpcs[Math.floor(Math.random() * activeNpcs.length)],
        motive: '推动剧情发展',
        mode: 'auto',
      };
    }
  }

  // 4. 默认旁白
  return { role: '旁白', motive: '承接剧情', mode: 'fast' };
}

// ========== Speaker Agent ==========
async function generateSpeech(speaker, motive, mode) {
  const cfg = getConfig();

  // T0 模板模式
  if (mode === 'fast' && speaker === '旁白') {
    return generateNarratorTemplate(motive);
  }

  // T1/T2 AI 生成
  const memory = cfg.memoryIntegration ? (tavo.get('tmm') || {}) : {};
  const prompt = buildSpeakerPrompt({
    speaker, motive, mode, memory,
    recentDialogue: tavo.get(`${NS}.context.dialogueHistory`) || [],
  });

  const raw = await tavo.generate(prompt, {
    context: false,
    settings: { temperature: 0.7 },
  });

  return parseSpeakerOutput(raw);
}

// ========== Orchestrator ==========
async function orchestrate() {
  if (orchestrating) return;
  orchestrating = true;

  try {
    const state = tavo.get(NS) || defaultState();
    const decision = decideSpeaker(state.context);

    const speech = await generateSpeech(decision.role, decision.motive, decision.mode);

    // 写入对话
    await tavo.message.append({
      content: speech,
      hidden: false,
    });

    // 更新状态
    state.context.currentSpeaker = decision.role;
    state.context.dialogueHistory.push({
      role: decision.role,
      content: speech,
      timestamp: Date.now(),
      speakerMode: decision.mode,
    });
    tavo.set(NS, state, 'chat');

  } catch (e) {
    console.warn('[mcs] orchestrate failed', e);
  } finally {
    orchestrating = false;
  }
}

// ========== Hooks ==========

// 初始化
tavo.plugin.on('chat:opened', async () => {
  const state = tavo.get(NS);
  if (!state || state.version !== 1) {
    tavo.set(NS, defaultState(), 'chat');
  }
  // 初始化角色列表
  await initActiveRoles();
});

// 用户发送消息
tavo.plugin.on('input:beforeSend', async (event) => {
  const text = event.text || '';
  // 解析 @角色 指令
  const atMatch = text.match(/@(\S+)/);
  if (atMatch) {
    event.targetRole = atMatch[1];
  }
});

// 消息添加
tavo.plugin.on('message:added', async (event) => {
  const cfg = getConfig();
  if (!cfg.enabled) return;

  const state = tavo.get(NS) || defaultState();

  // 记录对话
  state.context.dialogueHistory.push({
    role: event.message.characterName || event.message.role,
    content: event.message.content,
    timestamp: Date.now(),
    isUserMessage: event.message.role === 'user',
  });

  // 限制历史长度
  if (state.context.dialogueHistory.length > 20) {
    state.context.dialogueHistory = state.context.dialogueHistory.slice(-20);
  }

  // 更新计数
  if (event.message.role === 'user') {
    state.context.turnsSinceUserInput = 0;
  } else {
    state.context.turnsSinceUserInput++;
  }

  tavo.set(NS, state, 'chat');

  // 触发编排（用户消息后需要 NPC 回应）
  if (event.message.role === 'user') {
    await orchestrate();
  }
});

// 生成准备（注入上下文）
tavo.plugin.on('generation:prepare', async (event) => {
  const cfg = getConfig();
  if (!cfg.enabled) return;

  const state = tavo.get(NS);
  if (!state) return;

  // 注入编排上下文
  const inject = buildInjectionContext(state, cfg);
  if (inject) {
    event.text = inject + '\n\n---\n' + (event.text || '');
  }
});

// ========== Helpers ==========
async function initActiveRoles() {
  const chars = await tavo.character.find({});
  const state = tavo.get(NS) || defaultState();
  state.context.areaContext.activeRoles = chars.map(c => c.name);
  state.characters = {};
  for (const c of chars) {
    state.characters[c.id] = { name: c.name, type: c.roleType || 'npc', isKey: false };
  }
  tavo.set(NS, state, 'chat');
}
```

## 5. 实现优先级

| 优先级 | 功能 | 说明 |
|--------|------|------|
| P0 | 发言路由 | 决定谁说话 |
| P0 | 基础台词生成 | AI 生成台词 |
| P0 | 区域聚焦 | 控制谁可以发言 |
| P1 | NPC 自主发言 | 无用户输入时自动推进 |
| P1 | 记忆协同 | 与记忆插件联动 |
