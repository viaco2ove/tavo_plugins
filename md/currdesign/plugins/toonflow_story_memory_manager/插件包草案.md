## 4. entry.js 骨架

```js
// toonflow_story_memory_manager entry.js（骨架，省略具体实现体）

const NS = 'tmm';
let refreshing = false;

const SYSTEM_RULES = `（见 提示词设计.md 第 1 节，systemPromptOverride 非空时改用用户配置）`;

function getConfig() {
  return {
    enabled: tavo.plugin.config.get('enabled') !== false,
    payloadMode: tavo.plugin.config.get('payloadMode') || 'compact',
    refreshInterval: Number(tavo.plugin.config.get('refreshInterval') || 3),
    dialogueWindow: Number(tavo.plugin.config.get('dialogueWindow') || 6),
    factCap: Number(tavo.plugin.config.get('factCap') || 12),
    tagCap: Number(tavo.plugin.config.get('tagCap') || 8),
    injectEnabled: tavo.plugin.config.get('injectEnabled') !== false,
    injectBudget: Number(tavo.plugin.config.get('injectBudget') || 400),
    syncHostMemory: !!tavo.plugin.config.get('syncHostMemory'),
    snapshotEnabled: !!tavo.plugin.config.get('snapshotEnabled'),
    triggerKeywords: String(tavo.plugin.config.get('triggerKeywords') || '').split(/;|；/).map(s => s.trim()).filter(Boolean),
    worldSetting: String(tavo.plugin.config.get('worldSetting') || ''),
    systemPromptOverride: String(tavo.plugin.config.get('systemPromptOverride') || ''),
  };
}

// ---------- 初始化 ----------
tavo.plugin.on('chat:opened', async () => {
  const state = tavo.get(NS);
  if (!state || state.version !== 1) {
    tavo.set(NS, defaultState(), 'chat');
  }
});

// ---------- @记忆管理 直接指令 ----------
tavo.plugin.on('input:beforeSend', async (event) => {
  const cfg = getConfig();
  if (!cfg.enabled) return;
  const text = (event.text || '').trim();
  if (!text.startsWith('@记忆管理')) return;
  event.cancel();                    // 先取消发送，不受 5 秒限制
  runDirective(text).catch(err => console.warn('[tmm] directive failed', err));
});

async function runDirective(text) {
  const cfg = getConfig();
  const userPrompt = buildDirectivePrompt(text);   // 见 提示词设计.md 2.4
  const raw = await tavo.generate(systemHead(cfg) + '\n\n' + userPrompt, {
    context: false,
    settings: { temperature: 0.3 },
  });
  const result = parseMemoryResult(raw);
  if (result) {
    applyMemoryResult(result, cfg);
    tavo.utils.toast(tavo.plugin.i18n.t('runtime.directiveDone'));
  } else {
    tavo.utils.toast(tavo.plugin.i18n.t('runtime.directiveFailed', { error: '解析失败' }));
  }
}

// ---------- 异步记忆刷新（写入侧） ----------
tavo.plugin.on('message:added', async (event) => {
  const cfg = getConfig();
  if (!cfg.enabled || refreshing) return;
  if (event.message?.role === 'system') return;

  const meta = tavo.get(`${NS}.meta`) || defaultState().meta;
  meta.turnsSinceRefresh = (meta.turnsSinceRefresh || 0) + 1;

  const deltaText = String(event.message?.content || '');
  const hardTrigger = cfg.triggerKeywords.some(k => deltaText.includes(k));
  if (!hardTrigger && meta.turnsSinceRefresh < cfg.refreshInterval) {
    tavo.update(NS, { meta }, 'chat');
    return;
  }
  refreshing = true;
  try {
    await refreshMemory(cfg);      // 内部采集→generate→parse→apply；失败仅 warn
  } finally {
    refreshing = false;
  }
});

async function refreshMemory(cfg) {
  const chat = await tavo.chat.current();
  const state = tavo.get(NS) || defaultState();
  const count = await tavo.message.count();
  const recentMsgs = await tavo.message.find([-cfg.dialogueWindow, -1]);
  const deltaMsgs = await tavo.message.find([state.meta.lastIndex, -1]);
  const payload = buildMemoryPayload({ chat, state, recentMsgs, deltaMsgs, cfg });
  const userPrompt = buildMemoryUserPrompt(payload, cfg.payloadMode === 'compact');
  const raw = await tavo.generate(systemHead(cfg) + '\n\n' + userPrompt, {
    context: false,
    settings: { temperature: 0.3 },
  });
  const result = parseMemoryResult(raw);
  if (!result) { console.warn('[tmm] parse failed, keep old memory'); return; }
  applyMemoryResult(result, cfg, count);
  if (cfg.snapshotEnabled) await saveSnapshot(state);
}

// ---------- 生成时注入（读取侧） ----------
tavo.plugin.on('generation:prepare', async (event) => {
  const cfg = getConfig();
  if (!cfg.enabled || !cfg.injectEnabled) return;
  const state = tavo.get(NS);
  if (!state || !state.meta?.summary) return;
  const block = buildInjectionBlock(state, cfg.injectBudget);  // 见 提示词设计.md 第 5 节
  if (block) event.text = block + '\n\n---\n' + (event.text || '');
});

// ---------- 侧边栏动作 ----------
tavo.plugin.onSidebarAction('refresh-memory', async () => {
  if (refreshing) return tavo.utils.toast(tavo.plugin.i18n.t('runtime.refreshBusy'));
  refreshing = true;
  try { await refreshMemory(getConfig()); tavo.utils.toast(tavo.plugin.i18n.t('runtime.refreshDone')); }
  catch (e) { console.warn('[tmm] manual refresh failed', e); }
  finally { refreshing = false; }
});

tavo.plugin.onSidebarAction('inspect-memory', async () => {
  const state = tavo.get(NS);
  await tavo.message.append({ content: '```\n' + JSON.stringify(state, null, 2) + '\n```', hidden: true });
});

tavo.plugin.onSidebarAction('export-memory', async () => {
  const state = tavo.get(NS);
  await tavo.file.export(tavo.plugin.i18n.t('runtime.exportName'), JSON.stringify(state, null, 2));
});

tavo.plugin.onSidebarAction('writeback-cards', async () => {
  // 逐角色把 tmm.cards 写回角色卡（每次 tavo.character.update 都会弹用户确认）
  // 实现要点：cards.player → persona 不写回；cards.npcs 按 name 查 tavo.character.find
});

// ---------- 工具函数（实现时补齐） ----------
function defaultState() { /* 见 设计文档.md 第 5 节 */ }
function systemHead(cfg) { return cfg.systemPromptOverride || SYSTEM_RULES; }
function buildMemoryPayload(args) { /* 采集并截断，见 提示词设计.md 2.1 */ }
function buildMemoryUserPrompt(payload, compactMode) { /* 模板见 提示词设计.md 第 2 节 */ }
function buildDirectivePrompt(text) { /* 模板见 提示词设计.md 2.4 */ }
```

## 5. ui/panel.html 要点

- `position: fixed` 悬浮面板，**必须支持拖动**（鼠标 + 触摸），区分点击与拖动；
- 位置按 `tmm.panelPos`（chat 作用域）持久化，窗口变化后收回可视区；提供「重置位置」；
- 初始位置避开输入框与发送按钮（建议右下角偏上）；
- 展示 summary / facts / tags / 玩家状态行 + 手动刷新按钮；
- 首次按 `tavo.plugin.i18n` 渲染，并订阅 `i18n.onChange` 重渲染。
